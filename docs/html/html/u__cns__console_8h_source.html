132 void mk_blank(const char* cl_name, const char* desc, blank_t::callback_t foo){
133
134 if (!strcmp(cl_name, "help") || !strcmp(cl_name, "exit"))
135 {
136 return;
137 }
138
139 auto it = _blanks.find(cl_name);
140 if (it != _blanks.end())
141 {
142 _blanks.erase(it); // for user redefinition!
143 }
144
145 // just return
146 if (_blanks.size() > max_blanks)
147 {
148 return;
149 }
150
151 _blanks.emplace(cl_name, blank_t(desc, foo));
152 }

174 void ctask(ufo::token_t token){
175 if (_clb_pre)
176 {
177 bool r = _clb_pre();
178 if (!r)
179 {
180 return;
181 }
182 }
183
184 create_internal_blanks();
185
186 _blanks.find("help")->second.run(_block);
187
188 while (token && _block.get_event() != event_t::exit)
189 {
190 _block.write("u_cns>");
191 auto line = _block.read_line();
192 if (line.empty())
193 {
194 continue;
195 }
196
197 if (line.size())
198 {
199 find_and_go(line);
200 }
201 ufo::utl::sleep_for(50);
202 }
203 _block.write("Catched exit command\nConcole closed\n");
204
205 if (_clb_post)
206 {
207 _clb_post();
208 }
209 }

212 void create_internal_blanks() {
213
214 if (_blanks.find("exit") != _blanks.end())
215 {
216 _blanks.emplace("exit", blank_t("Command for exit from console", exit_cns));

226 bl << "Available commands (" << context.size() << " total):\n";

264 void find_and_go(std::string& s)
265 {
266 if (_blanks.empty())
267 {
268 return;
269 }
270
271 _block._parser.parse(s, true);
272 if (!_block._parser.good())
273 {
274 _block << "u_cns: bad arguments in line\n";
275 return;
276 }
277
278 auto blank = _block._parser.get_prog_name();
279
280 auto bl_it = _blanks.find(blank);
281 if (bl_it != _blanks.end())
282 {
283 bl_it->second.run(_block);
284 return;
285 }
286 _block << blank << ": command not found\n";
287 }
