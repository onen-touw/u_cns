458 {
459 opt_ind_t sind = {};
460
461 auto iter = begin;
462
463 // class
464 while (iter != end)
465 {
466 if (*iter == separator)

468 sind.cls = {begin, iter};

474

477 {
478 sind.cls = {begin, iter};
